program      ::= extern_list decl_list

extern_list  ::= extern extern_list
               | epsilon

extern       ::= "extern" type_spec IDENT "(" params ")" ";"

decl_list    ::= decl decl_list'
decl_list'   ::= decl decl_list'
               | epsilon

decl         ::= "void" IDENT "(" params ")" block
               | var_type IDENT decl_tail
decl_tail    ::= ";"
               | "(" params ")" block

type_spec    ::= "void"
               | var_type

var_type     ::= "int" | "float" | "bool"

params       ::= param_list
               | "void"
               | epsilon

param_list   ::= param param_list'
param_list'  ::= "," param param_list'
               | epsilon

param        ::= var_type IDENT

block        ::= "{" local_decls stmt_list "}"

local_decls  ::= local_decl local_decls
               | epsilon

local_decl   ::= var_type IDENT ";"

stmt_list    ::= stmt stmt_list
               | epsilon

stmt         ::= expr_stmt
               | block
               | if_stmt
               | while_stmt
               | return_stmt

expr_stmt    ::= expr ";"
               | ";"

while_stmt   ::= "while" "(" expr ")" stmt

if_stmt      ::= "if" "(" expr ")" block else_stmt

else_stmt    ::= "else" block
               | epsilon

return_stmt  ::= "return" return_tail
return_tail  ::= ";"
               | expr ";"

expr         ::= IDENT "=" expr
               | rval

rval         ::= and_expr rval'
rval'        ::= "||" and_expr rval'
               | epsilon

and_expr     ::= eq_expr and_expr'
and_expr'    ::= "&&" eq_expr and_expr'
               | epsilon

eq_expr      ::= rel_expr eq_expr'
eq_expr'     ::= "==" rel_expr eq_expr'
               | "!=" rel_expr eq_expr'
               | epsilon

rel_expr     ::= add_expr rel_expr'
rel_expr'    ::= "<=" add_expr rel_expr'
               | "<"  add_expr rel_expr'
               | ">=" add_expr rel_expr'
               | ">"  add_expr rel_expr'
               | epsilon

add_expr     ::= mul_expr add_expr'
add_expr'    ::= "+" mul_expr add_expr'
               | "-" mul_expr add_expr'
               | epsilon

mul_expr     ::= unary_expr mul_expr'
mul_expr'    ::= "*" unary_expr mul_expr'
               | "/" unary_expr mul_expr'
               | "%" unary_expr mul_expr'
               | epsilon

unary_expr   ::= "-" unary_expr
               | "!" unary_expr
               | primary

primary      ::= "(" expr ")"
               | IDENT ident_tail
               | INT_LIT
               | FLOAT_LIT
               | BOOL_LIT

ident_tail   ::= "(" args ")"
               | epsilon

args         ::= arg_list
               | epsilon

arg_list     ::= expr arg_list'
arg_list'    ::= "," expr arg_list'
               | epsilon
